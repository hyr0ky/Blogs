```java
import java.io.*;
import org.apache.commons.codec.binary.Base64;
import java.util.HashSet;
import java.util.Set;

public class Secure {

    // 安全的ObjectInputStream，只允许特定类
    static class SafeObjectInputStream extends ObjectInputStream {
        private final Set<String> allowedClasses = new HashSet<>();
        
        public SafeObjectInputStream(InputStream in) throws IOException {
            super(in);
            // 只允许User类及其依赖的基础类
            allowedClasses.add("com.example.User");
            allowedClasses.add("java.lang.String");
            allowedClasses.add("java.util.Date");
            // 添加其他必要的类...
        }
        
        @Override
        protected Class<?> resolveClass(ObjectStreamClass desc)
                throws IOException, ClassNotFoundException {
            String className = desc.getName();
            
            // 检查类是否在白名单中
            if (!isAllowed(className)) {
                throw new InvalidClassException(
                    "Unauthorized deserialization attempt", 
                    className
                );
            }
            
            return super.resolveClass(desc);
        }
        
        private boolean isAllowed(String className) {
            // 检查类名是否在白名单中
            for (String allowed : allowedClasses) {
                if (className.startsWith(allowed) || 
                    className.equals(allowed)) {
                    return true;
                }
            }
            return false;
        }
    }
    
    public User userFromString(String str) {
        try {
            // 1. Base64解码
            byte[] decoded = Base64.decodeBase64(str);
            ByteArrayInputStream in = new ByteArrayInputStream(decoded);
            
            // 2. 使用安全的ObjectInputStream
            try (SafeObjectInputStream iin = new SafeObjectInputStream(in)) {
                // 3. 反序列化
                Object obj = iin.readObject();
                
                // 4. 类型检查
                if (obj instanceof User) {
                    return (User) obj;
                } else {
                    throw new IllegalArgumentException(
                        "Deserialized object is not a User"
                    );
                }
            }
        } catch (Exception e) {
            // 5. 安全地记录异常（不泄露内部信息）
            logError("反序列化失败", e);
            return null;
        }
    }
    
    private static void logError(String message, Exception e) {
        // 记录日志但不暴露堆栈给用户
        System.err.println(message + ": " + e.getMessage());
        // 生产环境应该使用日志框架
    }
}
```

```python
# 导入Python标准库中的HTTP服务器模块和Cookie处理模块
from http.server import BaseHTTPRequestHandler, HTTPServer
from http.cookies import SimpleCookie

# 自定义HTTP请求处理类，继承自BaseHTTPRequestHandler
class MyServer(BaseHTTPRequestHandler):
    
    # 处理客户端GET请求的方法
    def do_GET(self):
        """
        处理HTTP GET请求的核心方法。
        从请求的Cookie中提取session_id，并尝试从文件中读取对应的用户名。
        如果找不到session_id，则使用默认用户名'stranger'。
        最后返回一个简单的HTML页面问候用户。
        """
        
        # 从HTTP请求头中获取Cookie字符串，并解析为SimpleCookie对象
        cookies = SimpleCookie(self.headers.get('Cookie'))
        
        # 检查Cookie中是否存在名为'session_id'的项
        if cookies.get('session_id'):
            # 【严重安全漏洞】路径遍历漏洞
            # 1. 直接使用客户端提供的session_id值作为文件名
            # 2. 没有对文件名进行任何验证或清理
            # 3. 攻击者可以传入如"../../../etc/passwd"这样的路径来读取系统文件
            # 4. 示例攻击：Cookie: session_id=../../etc/passwd
            username = open(cookies.get('session_id').value).readlines()[0]
        else:
            # 如果Cookie中没有session_id，使用默认用户名
            username = 'stranger'
        
        # 发送HTTP响应状态码200（成功）
        self.send_response(200)
        # 设置响应头，指定内容类型为HTML
        self.send_header("Content-type", "text/html")
        # 结束响应头部的设置
        self.end_headers()
        
        # 开始构建并发送HTML响应体
        # 注意：这里的用户名直接插入HTML中，存在XSS跨站脚本攻击漏洞
        # 如果用户名包含HTML或JavaScript代码，会被浏览器执行
        self.wfile.write(bytes("<html><head><title>Hello</title></head>", "utf-8"))
        self.wfile.write(bytes("<body>", "utf-8"))
        
        # 【安全漏洞】XSS跨站脚本攻击
        # username变量未经任何转义直接插入HTML，如果username包含恶意脚本会被执行
        # 例如：username = "<script>alert('XSS')</script>"
        self.wfile.write(bytes("<h1>Hello %s</h1>" % username, "utf-8"))
        
        self.wfile.write(bytes("</body></html>", "utf-8"))

# 程序入口点
if __name__ == "__main__":
    """
    启动HTTP服务器的主程序部分。
    服务器监听所有网络接口（0.0.0.0）的1337端口。
    """
    
    # 创建HTTPServer实例
    # 参数说明：
    # - ('0.0.0.0', 1337): 监听所有网络接口的1337端口
    #   【安全风险】0.0.0.0表示公开访问，生产环境应谨慎使用
    # - MyServer: 上面定义的自定义请求处理类
    webServer = HTTPServer(('0.0.0.0', 1337), MyServer)
    
    try:
        # 启动服务器，使其持续运行并处理请求
        # serve_forever()方法会一直运行，直到程序被中断
        webServer.serve_forever()
    except KeyboardInterrupt:
        # 捕获键盘中断（Ctrl+C），优雅地停止服务器
        pass
    
    # 关闭服务器套接字，释放资源
    webServer.server_close()
    # 输出服务器已停止的消息
    print("Server stopped.")
```
